cmake_minimum_required(VERSION 3.16)
project(what-changed VERSION 1.0.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CTest)
option(WHAT_CHANGED_SANITIZERS "Enable address/undefined sanitizers" OFF)

add_library(what_changed_lib
  src/json.cpp src/util.cpp src/snapshot.cpp src/collectors.cpp src/diff.cpp src/output.cpp)
target_include_directories(what_changed_lib PUBLIC include)
target_compile_features(what_changed_lib PUBLIC cxx_std_17)
target_compile_definitions(what_changed_lib PUBLIC WHAT_CHANGED_VERSION="${PROJECT_VERSION}")
target_compile_options(what_changed_lib PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(WHAT_CHANGED_SANITIZERS)
  target_compile_options(what_changed_lib PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(what_changed_lib PUBLIC -fsanitize=address,undefined)
endif()

add_executable(what-changed src/main.cpp)
target_link_libraries(what-changed PRIVATE what_changed_lib)
target_compile_options(what-changed PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(BUILD_TESTING)
  add_executable(what-changed-tests tests/tests.cpp)
  target_link_libraries(what-changed-tests PRIVATE what_changed_lib)
  target_compile_options(what-changed-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME what-changed-tests COMMAND what-changed-tests)
endif()

install(TARGETS what-changed RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/what-changed.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)

set(CPACK_PACKAGE_NAME "what-changed")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_PACKAGE_VENDOR "GuhsHosting")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/guhshosting/what-changed")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Privacy-conscious Linux system change detector")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/guhshosting/what-changed")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6, libstdc++6")
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_GROUP "Applications/System")
set(CPACK_RPM_PACKAGE_VENDOR "GuhsHosting")
set(CPACK_RPM_PACKAGE_URL "https://github.com/guhshosting/what-changed")
include(CPack)
